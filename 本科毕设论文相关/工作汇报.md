# -*- coding: utf-8 -*-
import os
import mujoco

def urdf_to_mujoco(urdf_path, mjcf_path=None):
    """
    urdf_to_mujoco 的 Docstring
    
    :param urdf_path: URDF 文件路径
    :param mjcf_path: 保存MJCF文件路径，如果为空则默认在URDF文件所在目录
    :return: MJCF文件的路径
    """

    print("开始将URDF文件转换为MJCF文件...")
    print(f"URDF文件路径: {urdf_path}")

    #检查URDF文件是否存在
    if not os.path.exists(urdf_path):
        print(f"错误: URDF文件不存在 - {urdf_path}")
        return None

    #设置默认的MJCF保存路径
    if mjcf_path is None:
        urdf_dir =os.path.dirname(urdf_path)
        urdf_name = os.path.splitext(os.path.basename(urdf_path))[0]
        mjcf_path = os.path.join(urdf_dir,f"{urdf_name}.xml")

    try:
        # 1. 加载URDF文件（MUJOCO会自动将其编译为内部表示）
        model = mujoco.MjModel.from_xml_path(urdf_path)

        # 2. 创建MjData对象（保存函数需要）
        data = mujoco.MjData(model)

        # 3. 保存为MJCF格式
        mujoco.mj_saveLastXML(mjcf_path,model)

        print(f" MJCF文件保存成功: {mjcf_path}")
        print(f" 模型信息:")
        print(f"  - 关节数量: {model.njnt}")
        print(f"  - 连杆数量: {model.nbody}")

        return mjcf_path
    except Exception as e:
        print(f"转换过程中出现错误: {e}")
        import traceback
        traceback.print_exc()
        return None

def main():
    print("URDF转MJCF工具启动")

    urdf_file = "e:\\apps\\mujoco\\mujoco_study_wu\\mujoco_learning-main\\mujoco_learning_wu\\3dof_arm.urdf"
    mjcf_file = "e:\\apps\\mujoco\\mujoco_study_wu\\mujoco_learning-main\\mujoco_learning_wu\\3dof_arm_mjcf.xml" 

    result = urdf_to_mujoco(urdf_file, mjcf_file) 

    if result:
        print("转换完成！你可以使用以下命令加载MJCF模型：")
        print(f"   python -c \"import mujoco; import mujoco.viewer; model = mujoco.MjModel.from_xml_path('{mjcf_file.replace(chr(92), chr(92)*2)}'); data = mujoco.MjData(model); mujoco.viewer.launch(model, data)\"")
    else:
        print("转换失败，请检查错误信息。")
    
if __name__ == "__main__":
    main()

##  激励轨迹
在 MuJoCo 的 Python 接口中让机械臂跟踪最优激励轨迹的核心逻辑是：在模拟的每一时间步，根据当前模拟时间计算 / 查询轨迹对应的关节指令（角度 / 速度 / 力矩），并赋值给 MuJoCo 的控制变量（data.ctrl）。  
1. 执行器（actuator）映射到对应关节（joint）(位置/速度/力矩)  
2. 轨迹通常是[时间序列+关节指令序列]  
  -连续轨迹：如解析函数(q(t)=a*sin(wt)+b)
  -离散轨迹：时间-关节值列表（[(t0,q0),(t1,q1),..., (tn,qn)]）

### 定义/预处理最优激励轨迹
- 连续轨迹：直接定义时间函数
- 离散轨迹：插值(线性/三次采样)补全时间间隙，确保任意时刻能查询到关节指令

### 模拟循环中跟踪轨迹
在每个模拟步中  
1. 获取当前模拟时间（data.time）  
2. 根据当前时间查询期望关节指令;   
3. 将指令赋值给data.ctrl    
4. 执行模拟步（mj_step）  
5. 可视化/记录数据  



## 正弦波的激励轨迹_to_panda
```python
import mujoco as mj
import numpy as np
import matplotlib.pyplot as plt
from mujoco import viewer

# 1.加载panda模型&初始化
#加载xml模型
model = mj.MjModel.from_xml_path(r"E:\apps\mujoco\mujoco_study_wu\franka_emika_panda\panda.xml")
#创建数据容器，存储仿真状态
data = mj.MjData(model)

#读取home位姿的keyframe
#xml中keyframe “home”的qpos和ctrl
home_qpos = np.array([0 ,0 , 0, -1.57079, 0 ,1.57079 ,-0.7853, 0.04 ,0.04])
home_ctrl = np.array([0 ,0 ,0 ,-1.57079, 0, 1.57079, -0.7853, 255])

#初始化状态
data.qpos[:] = home_qpos
data.ctrl[:] = home_ctrl
mj.mj_forward(model,data) #前向计算初始化

#启动被动可视化窗口  
viewer_handle = viewer.launch_passive(model,data)

#2. 定义关节激励轨迹
total_time =10.0 #总时间(s)
dt = model.opt.timestep #仿真步长 默认0.002s
n_steps = int(total_time / dt) #总步数

#生成正弦激励轨迹（7个臂关节）
q_desired = np.zeros((n_steps,7)) #预分配目标位置数组
t_list = np.linspace(0,total_time,n_steps) #时间序列

#为每个关节定义不同频率/振幅的正弦轨迹
for joint_idx in range(7):
    freq = 0.5 +joint_idx*0.1 #频率递增
    amp = 0.2 +joint_idx*0.05 #振幅递增
    q_desired[:,joint_idx] = home_qpos[joint_idx] + amp * np.sin(2 * np.pi * freq * t_list) #目标位置

#夹爪保持张开
gripper_ctrl = 255

#3. pd控制器
#pd参数
Kp = np.array([800,800,600,600,400,400,200]) #位置增益 关节1-7比例增益
Kd = np.array([50,50,40,40,20,20,10])       #速度增益  关节1-7微分增益

#存储仿真数据
q_actual = np.zeros((n_steps,7)) #实际位置
tau_cmd = np.zeros((n_steps,7))  #控制力矩

#4. 仿真循环
try:
    for step in range(n_steps):
        #1. 读取当前关节状态
        q_current = data.qpos[:7] #前七个是臂关节角度
        dq_current = data.qvel[:7] #前七个是臂关节速度

        #2. 计算期望轨迹（位置+速度）
        q_d = q_desired[step] #当前时间步的目标位置
        dq_d = np.zeros(7) #目标速度设为0（或根据需求调整）
        if step < n_steps-1:
            dq_d = (q_desired[step+1]-q_desired[step]) / dt #轨迹导数
        
        # 3. pd控制律+重力补偿
        tau =Kp * (q_d - q_current) +Kd * (dq_d -dq_current)
        tau += data.qfrc_bias[:7]  #加入重力补偿项

        # 4. 限制力矩，匹配xml的forcerange
        tau = np.clip(tau, -87, 87)

        # 5. 赋值给控制输入
        data.ctrl[:7]=tau
        data.ctrl[7]=gripper_ctrl

        #6.步进仿真
        mj.mj_step(model,data)

        #7. 可视化更新,必须手动render
        viewer_handle.sync()

        #8. 记录数据
        q_actual[step]=q_current
        tau_cmd[step]=tau

finally:
    viewer_handle.close()

#5.轨迹跟踪结果可视化
plt.figure(figsize=(12,8))
for joint_idx in range(7):
    plt.subplot(7,1,joint_idx+1)
    plt.plot(t_list,q_desired[:,joint_idx],'r--',label="Desired")
    plt.plot(t_list,q_actual[:,joint_idx],'b-',label="Actual")
    plt.title(f"Joint {joint_idx+1} Angle")
    plt.ylabel("Angle (rad)")
    plt.legend()
plt.xlabel("Time (s)")
plt.tight_layout()
plt.show()
```

这里是没有加重力补偿项目，反而更契合了  
原因是：重力补偿项在 “动态轨迹跟踪” 场景下，对高增益 PD 控制器形成了 “过度补偿 / 额外干扰”  
data.qfrc_bias 是 MuJoCo 官方定义的关节偏置力矩，设计初衷是：抵消关节在当前位置 / 速度下的 “自然力矩”，让关节在无外部控制时能保持静止（比如机械臂举到某个位置，不加控制会因重力下落，加 qfrc_bias 就能抵消重力，保持静止）。这个补偿项是为静态位置保持设计的，而非动态轨迹跟踪。

# 5次多项式轨迹生成函数
def generate_fifth_order_polynomial(q0, qf, T, n_steps):
    """
    生成5次多项式轨迹
    q0: 起始位置
    qf: 结束位置
    T: 总时间
    n_steps: 总步数
    返回: 轨迹位置数组
    """
    t = np.linspace(0, T, n_steps)
    t_normalized = t / T
    
    # 计算5次多项式系数
    a0 = q0
    a1 = 0
    a2 = 0
    a3 = 10 * (qf - q0)
    a4 = -15 * (qf - q0)
    a5 = 6 * (qf - q0)
    
    # 计算位置轨迹
    q_t = a0 + a1 * t_normalized + a2 * t_normalized**2 + a3 * t_normalized**3 + a4 * t_normalized**4 + a5 * t_normalized**5
    
    return q_t

#生成5次多项式轨迹（7个臂关节）
q_desired = np.zeros((n_steps,7)) #预分配目标位置数组
t_list = np.linspace(0,total_time,n_steps) #时间序列

#为每个关节定义不同的5次多项式轨迹
for joint_idx in range(7):

    q2 = generate_fifth_order_polynomial(home_qpos[joint_idx] + q_offset, home_qpos[joint_idx], total_time/2, n_steps - n_steps_half)
    
    # 合并两段轨迹
    q_desired[:, joint_idx] = np.concatenate((q1, q2))

#夹爪保持张开
gripper_ctrl = 255 


#3. pd控制器 
#pd参数 
Kp = np.array([600,600,400,400,300,300,80]) #位置增益 关节1-7比例增益 
Kd = np.array([40,40,35,35,20,20,8])       #速度增益  关节1-7微分增益

q_actual = np.zeros((n_steps,7)) #实际位置 
tau_cmd = np.zeros((n_steps,7))   #控制力矩 


#4. 仿真循环 
try: 
    for step in range(n_steps): 
        #1. 读取当前关节状态

        dq_current = data.qvel[:7] #前七个是臂关节速度 

        #2. 计算期望轨迹（位置+速度） 
        q_d = q_desired[step] #当前时间步的目标位置 
        dq_d = np.zeros(7) #目标速度设为0（或根据需求调整） 
        if step < n_steps-1: 
            dq_d = (q_desired[step+1]-q_desired[step]) / dt #轨迹导数 
        
        # 3. pd控制律+重力补偿 
        tau = Kp * (q_d - q_current) + Kd * (dq_d - dq_current) 
        # tau += data.qfrc_bias[:7]  #加入重力补偿项 

        # 4. 限制力矩，匹配xml的forcerange 
        tau = np.clip(tau, -87, 87) 

        # 5. 赋值给控制输入 
        data.ctrl[:7] = tau 
        data.ctrl[7] = gripper_ctrl 

        #6.步进仿真 
        mj.mj_step(model,data) 

        #7. 可视化更新,必须手动render 
        viewer_handle.sync() 

        #8. 记录数据

        tau_cmd[step] = tau 

finally: 
    viewer_handle.close()

### urdf和mujoco xml(mjcf)的差异
MuJoCo XML（你的panda.xml）：使用MuJoCo特有的标签，如`<mujoco>、<compiler>、<option>、<tendon>、<actuator>`等

### mjcf转urdf(mjcf_urdf_simple_converter)
[mjcf_urdf_simple_converter](https://pypi.org/project/mjcf-urdf-simple-converter/)  
在URDF中，当关节连接父体和子体时，子体的原点固定在关节的位置[维基百科](https://wiki.ros.org/urdf/XML/joint)。  
连接父母和孩子之间始终只有一个连接点。另一方面，对于MJCFs，一个主体中可以定义多个关节，且在子体元素中定义了多个关节。  
此外，身体位置相对于母帧定义，不依赖于关节位置。因此，为了将MJCF的运动学结构转换为URDF是必要的。  

我在这段代码中采用的方法是添加许多中间体，这些中间体的相对姿态用关节定义，来表达每个变换。该函数在关节位置创建一个质量和惯性可忽略的物体，从而形成一个空物体，关节可以连接到其上。这样，MJCF中定义的每个实际天体都可以保持使用相同的坐标系fixedadd_dummy_body()

例如，如果MJCF中的一个物体有两个关节：

- 一个接点连接到revoluteparent_bodyjoint1_jointbody
- 一个接点连接到revolutejoint1_jointbodyjoint2_jointbody
- 一个关节连接到（这“将坐标系”带回“到MJCF子体的参考系）fixedjoint2_jointbodychild_body  
这一过程会继续到下一个MJCF机构。

```py
from mjcf_urdf_simple_converter import convert
convert("model.xml", "model.urdf")
# or, if you are using it in your ROS package and would like for the mesh directories to be resolved correctly, set meshfile_prefix, for example:
convert("model.xml", "model.urdf", asset_file_prefix="package://your_package_name/model/")
```

```py
# -*- coding: utf-8 -*-
import os
import argparse

# 导入指定的转换库
from mjcf_urdf_simple_converter import convert

def mjcf_to_urdf(mjcf_path, urdf_path=None, asset_file_prefix=None):
    """
    将MJCF文件转换为URDF文件
    
    :param mjcf_path: MJCF 文件路径
    :param urdf_path: 保存URDF文件路径，如果为空则默认在MJCF文件所在目录
    :param asset_file_prefix: 资产文件前缀，用于正确解析网格目录（可选）
    :return: URDF文件的路径
    """

    print("开始将MJCF文件转换为URDF文件...")
    print(f"MJCF文件路径: {mjcf_path}")

    #检查MJCF文件是否存在
    if not os.path.exists(mjcf_path):
        print(f"错误: MJCF文件不存在 - {mjcf_path}")
        return None

    #设置默认的URDF保存路径
    if urdf_path is None:
        mjcf_dir = os.path.dirname(mjcf_path)
        mjcf_name = os.path.splitext(os.path.basename(mjcf_path))[0]
        urdf_path = os.path.join(mjcf_dir, f"{mjcf_name}.urdf")

    try:
        # 使用指定的转换库转换MJCF到URDF
        convert(mjcf_path, urdf_path, asset_file_prefix=asset_file_prefix)

        print(f"URDF文件保存成功: {urdf_path}")
        print(f"转换完成！")

        return urdf_path
    except Exception as e:
        print(f"转换过程中出现错误: {e}")
        import traceback
        traceback.print_exc()
        return None

def main():
    print("MJCF转URDF工具启动")
    
    # 创建参数解析器
    parser = argparse.ArgumentParser(description='将MJCF文件转换为URDF文件')
    
    # 将mjcf_path设为必需参数，不提供默认值
    parser.add_argument('mjcf_path', type=str, help='MJCF文件路径（必需）')
    
    # 将urdf_path设为可选参数
    parser.add_argument('--urdf_path', '-o', type=str, help='保存URDF文件路径（可选，默认与MJCF同目录）')
    
    # 添加asset_file_prefix参数
    parser.add_argument('--asset_file_prefix', type=str, help='资产文件前缀，用于正确解析网格目录（可选）')
    
    # 解析命令行参数
    args = parser.parse_args()
    
    # 如果用户只提供了位置参数mjcf_path，args.urdf_path会是None
    result = mjcf_to_urdf(args.mjcf_path, args.urdf_path, args.asset_file_prefix) 

    if result:
        print("转换完成！你可以使用以下命令查看URDF模型：")
        print(f"   python -c \"from urdfpy import URDF; robot = URDF.load('{result.replace(chr(92), chr(92)*2)}'); print(robot.joints)\"")
    else:
        print("转换失败，请检查错误信息。")
    
if __name__ == "__main__":
    main()

```
